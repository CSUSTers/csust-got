# Go 1.27 and Paimon Migration Implementation Plan

> **For agentic workers:** Use the subagent-driven-development skill to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate every active Go build surface to Go 1.27 and prove that the existing Sonic import path is supplied by Paimon v0.1.0 with Eino-compatible JSON semantics and a non-race performance ceiling of 2.5× Go 1.27 `encoding/json`.

**Architecture:** Keep all production and upstream imports at `github.com/bytedance/sonic`, select the released Paimon module through one root-module `replace`, and lock that selection with build-info and semantic compatibility tests in `chatv2`. Keep functional/race verification separate from a same-process, non-race performance gate; wire the latter through an explicit Make target and an independent CI step so `make test` cannot silently exclude it.

**Tech Stack:** Go 1.27, Go modules, `github.com/hugefiver/paimon-go v0.1.0` behind the Sonic module identity, CloudWeGo Eino schema types, `testing.Benchmark`, Testify, Make, GitHub Actions, Travis CI, golangci-lint v2 configuration, Docker/Buildah, Dev Containers, and PowerShell 7.

  ```go
  //go:build !race && !386 && !arm

  package chatv2

  import (
	"encoding/json"
	"testing"

	"github.com/bytedance/sonic"
	"github.com/stretchr/testify/require"
  )

  const maxJSONBackendSlowdown = 2.5

  var (
	jsonBackendBytesSink  []byte
	jsonBackendValueSink  jsonBackendRepresentativePayload
	jsonBackendStringSink string
	jsonBackendBoolSink   bool
  )

  type jsonBackendMarshalFunc func(any) ([]byte, error)
  type jsonBackendUnmarshalFunc func([]byte, any) error

  func runJSONBackendMarshalBenchmark(b *testing.B, payload jsonBackendRepresentativePayload, marshal jsonBackendMarshalFunc) {
	b.Helper()
	b.ReportAllocs()
	for b.Loop() {
		data, err := marshal(payload)
		if err != nil {
			b.Fatal(err)
		}
		jsonBackendBytesSink = data
	}
  }

  func runJSONBackendUnmarshalBenchmark(b *testing.B, data []byte, unmarshal jsonBackendUnmarshalFunc) {
	b.Helper()
	b.ReportAllocs()
	for b.Loop() {
		var value jsonBackendRepresentativePayload
		if err := unmarshal(data, &value); err != nil {
			b.Fatal(err)
		}
		jsonBackendValueSink = value
	}
  }

  func benchmarkJSONBackendMarshal(payload jsonBackendRepresentativePayload, marshal jsonBackendMarshalFunc) testing.BenchmarkResult {
	return testing.Benchmark(func(b *testing.B) {
		runJSONBackendMarshalBenchmark(b, payload, marshal)
	})
  }

  func benchmarkJSONBackendUnmarshal(data []byte, unmarshal jsonBackendUnmarshalFunc) testing.BenchmarkResult {
	return testing.Benchmark(func(b *testing.B) {
		runJSONBackendUnmarshalBenchmark(b, data, unmarshal)
	})
  }

  func requireJSONBackendWithinRatio(t *testing.T, name string, paimon, standard testing.BenchmarkResult) {
	t.Helper()
	t.Logf("%s: paimon=%d ns/op %d B/op %d allocs/op; encoding/json=%d ns/op %d B/op %d allocs/op",
		name,
		paimon.NsPerOp(), paimon.AllocedBytesPerOp(), paimon.AllocsPerOp(),
		standard.NsPerOp(), standard.AllocedBytesPerOp(), standard.AllocsPerOp())
	require.LessOrEqual(t, float64(paimon.NsPerOp()), float64(standard.NsPerOp())*maxJSONBackendSlowdown,
		"Paimon %s exceeded %.1fx encoding/json", name, maxJSONBackendSlowdown)
  }

  func TestJSONBackendPerformanceGate(t *testing.T) {
	payload := newJSONBackendRepresentativePayload(t)
	encoded, err := json.Marshal(payload)
	require.NoError(t, err)

	standardMarshal := benchmarkJSONBackendMarshal(payload, json.Marshal)
	paimonMarshal := benchmarkJSONBackendMarshal(payload, sonic.Marshal)
	requireJSONBackendWithinRatio(t, "marshal", paimonMarshal, standardMarshal)

	standardUnmarshal := benchmarkJSONBackendUnmarshal(encoded, json.Unmarshal)
	paimonUnmarshal := benchmarkJSONBackendUnmarshal(encoded, sonic.Unmarshal)
	requireJSONBackendWithinRatio(t, "unmarshal", paimonUnmarshal, standardUnmarshal)
  }

  func BenchmarkJSONBackend(b *testing.B) {
	payload := newJSONBackendRepresentativePayload(b)
	encoded, err := json.Marshal(payload)
	if err != nil {
		b.Fatal(err)
	}
	encodedString := string(encoded)

	b.Run("Marshal/Paimon", func(b *testing.B) {
		runJSONBackendMarshalBenchmark(b, payload, sonic.Marshal)
	})
	b.Run("Marshal/EncodingJSON", func(b *testing.B) {
		runJSONBackendMarshalBenchmark(b, payload, json.Marshal)
	})
	b.Run("Unmarshal/Paimon", func(b *testing.B) {
		runJSONBackendUnmarshalBenchmark(b, encoded, sonic.Unmarshal)
	})
	b.Run("Unmarshal/EncodingJSON", func(b *testing.B) {
		runJSONBackendUnmarshalBenchmark(b, encoded, json.Unmarshal)
	})
	b.Run("GetFromString/Paimon", func(b *testing.B) {
		b.ReportAllocs()
		for b.Loop() {
			node, getErr := sonic.GetFromString(encodedString, "messages", 0, "tool_calls", 0, "function", "arguments")
			if getErr != nil {
				b.Fatal(getErr)
			}
			jsonBackendStringSink, getErr = node.String()
			if getErr != nil {
				b.Fatal(getErr)
			}
		}
	})
	b.Run("Valid/Paimon", func(b *testing.B) {
		b.ReportAllocs()
		for b.Loop() {
			jsonBackendBoolSink = sonic.Valid(encoded)
		}
	})
  }
  ```
